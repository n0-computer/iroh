21/// The Prefix/L of all Unique Local Addresses.
22const ADDR_PREFIXL: u8 = 0xfd;

25const ADDR_GLOBAL_ID: [u8; 5] = [0x15, 0x07, 0x0a, 0x51, 0x0b];

33/// The Subnet ID for [`EndpointIdMappedAddr`]: fd15:70a:510b::/64.
34const ENDPOINT_ID_SUBNET: [u8; 2] = [0x00, 0x00];

59/// Generic mapped address.
60///
61/// Allows implementing [`AddrMap`].
62pub(crate) trait MappedAddr {
63    /// Generates a new mapped address in the IPv6 Unique Local Address space.
64    ///
65    /// The host bits are random so a remote peer cannot guess one of our live mapped
66    /// addresses (uniqueness is ensured by [`AddrMap::get`]).
67    fn generate() -> Self;
68
69    /// Returns a consistent [`SocketAddr`] for the mapped addr.
70    ///
71    /// This socket address does not have a routable IP address.  It uses a fake but
72    /// consistent port number, since the port does not play a role in the addressing.  This
73    /// socket address is only to be used to pass into Noq.
74    fn private_socket_addr(&self) -> SocketAddr;
75}

77/// An enum encompassing all the mapped and unmapped addresses.
78///
79/// This is essentially a slightly-stronger typed version of the IPv6 mapped addresses that
80/// we use on the Noq side.  It categorises the addressed in what kind of mapped or
81/// unmapped addresses they are.
82///
83/// It does not guarantee that a mapped address exists in the mapping.  Or that a particular
84/// address is even supported on this platform.  Hence no wasm exceptions here.
85#[derive(Clone, Debug)]
86pub(crate) enum MultipathMappedAddr {
87    /// An address for a [`crate::EndpointId`], via one or more paths.
88    Mixed(EndpointIdMappedAddr),
89    /// An address for a particular [`crate::EndpointId`] via a particular relay.
90    Relay(RelayMappedAddr),
91    /// An IP based transport address.
92    Ip(SocketAddr),
93    /// Custom transport address.
94    Custom(CustomMappedAddr),
95}
96
97impl From<SocketAddr> for MultipathMappedAddr {
98    fn from(value: SocketAddr) -> Self {
99        match value.ip() {
100            IpAddr::V4(_) => Self::Ip(value),
101            IpAddr::V6(addr) => {
102                if let Ok(addr) = EndpointIdMappedAddr::try_from(addr) {
103                    return Self::Mixed(addr);
104                }
105                if let Ok(addr) = RelayMappedAddr::try_from(addr) {
106                    return Self::Relay(addr);
107                }
108                if let Ok(addr) = CustomMappedAddr::try_from(addr) {
109                    return Self::Custom(addr);
110                }
111                Self::Ip(value)
112            }
113        }
114    }
115}

117/// An address used to address a endpoint on any or all paths.
118///
119/// This is only used for initially connecting to a remote endpoint.  We instruct Noq to
120/// send to this address, and duplicate all packets for this address to send on all paths we
121/// might want to send the initial on:
122///
123/// - If this the first connection to the remote endpoint we don't know which path will work
124///   and send to all of them.
125///
126/// - If there already is an active connection to this endpoint we now which path to use.
127///
128/// It is but a newtype around an IPv6 Unique Local Addr.  And in our QUIC-facing socket
129/// APIs like [`noq::AsyncUdpSocket`] it comes in as the inner [`Ipv6Addr`], in those
130/// interfaces we have to be careful to do the conversion to this type.
131#[derive(Debug, Copy, Clone, PartialEq, Eq, Hash)]
132pub(crate) struct EndpointIdMappedAddr(Ipv6Addr);
133
134impl MappedAddr for EndpointIdMappedAddr {
135    /// Generates a globally unique fake UDP address.
136    ///
137    /// This generates and IPv6 Unique Local Address according to RFC 4193.
138    fn generate() -> Self {
139        let mut addr = [0u8; 16];
140        addr[0] = ADDR_PREFIXL;
141        addr[1..6].copy_from_slice(&ADDR_GLOBAL_ID);
142        addr[6..8].copy_from_slice(&ENDPOINT_ID_SUBNET);
143        rand::rng().fill_bytes(&mut addr[8..16]);
144
145        Self(Ipv6Addr::from(addr))
146    }
147
148    /// Returns a consistent [`SocketAddr`] for the [`EndpointIdMappedAddr`].
149    ///
150    /// This socket address does not have a routable IP address and port.
151    ///
152    /// This uses a made-up port number, since the port does not play a role in the
153    /// addressing.  This socket address is only to be used to pass into Noq.
154    fn private_socket_addr(&self) -> SocketAddr {
155        SocketAddr::new(IpAddr::from(self.0), MAPPED_PORT)
156    }
157}

161        write!(f, "EndpointIdMappedAddr({})", self.0)
162    }

165impl TryFrom<Ipv6Addr> for EndpointIdMappedAddr {
166    type Error = EndpointIdMappedAddrError;
167
168    fn try_from(value: Ipv6Addr) -> Result<Self, Self::Error> {
169        let octets = value.octets();
170        if octets[0] == ADDR_PREFIXL
171            && octets[1..6] == ADDR_GLOBAL_ID
172            && octets[6..8] == ENDPOINT_ID_SUBNET
173        {
174            return Ok(Self(value));
175        }
176        Err(e!(EndpointIdMappedAddrError))
177    }

317/// A bi-directional map between a key and a [`MappedAddr`].
318///
319/// Entries are only removed explicitly with [`AddrMap::remove`]. Keys come from trusted
320/// sources or are removed by their owner, see [`RelayAddrMap`] for relay-chosen keys.
321///
322/// The forward map uses the standard library's randomly seeded hasher because keys may be
323/// chosen by remotes whereas the reverse map uses `FxHashMap` because mapped addresses are
324/// generated locally at random.
325#[derive(Debug, Clone)]
326pub(super) struct AddrMap<K, V> {
327    inner: Arc<std::sync::Mutex<AddrMapInner<K, V>>>,
328}
329
330// Manual impl because derive ends up requiring T: Default.
331impl<K, V> Default for AddrMap<K, V> {
332    fn default() -> Self {
333        Self {
334            inner: Default::default(),
335        }
336    }
337}
338
339impl<K, V> AddrMap<K, V>
340where
341    K: Eq + Hash + Clone + fmt::Debug,
342    V: MappedAddr + Eq + Hash + Copy + fmt::Debug,
343{
344    /// Returns the [`MappedAddr`], generating one if needed.
345    pub(super) fn get(&self, key: &K) -> V {
346        let mut inner = self.inner.lock().expect("poisoned");
347        match inner.addrs.get(key) {
348            Some(addr) => *addr,
349            None => {
350                let addr = generate_unique(&inner.lookup);
351                inner.addrs.insert(key.clone(), addr);
352                inner.lookup.insert(addr, key.clone());
353                trace!(?addr, ?key, "generated new addr");
354                addr
355            }
356        }
357    }
358
359    /// Removes the entry for `key`, if any.
360    pub(super) fn remove(&self, key: &K) {
361        let mut inner = self.inner.lock().expect("poisoned");
362        if let Some(addr) = inner.addrs.remove(key) {
363            inner.lookup.remove(&addr);
364            trace!(?addr, ?key, "removed addr");
365        }
366    }
367
368    /// Performs the reverse lookup.
369    pub(super) fn lookup(&self, addr: &V) -> Option<K> {
370        let inner = self.inner.lock().expect("poisoned");
371        inner.lookup.get(addr).cloned()
372    }
373}

391/// Generates a mapped address not yet present in `lookup`.
392///
393/// It's important these are generated randomly and not a counter in order to be
394/// unpredictable from the outside.
395fn generate_unique<V: MappedAddr + Eq + Hash, K>(lookup: &FxHashMap<V, K>) -> V {
396    loop {
397        let candidate = V::generate();
398        if !lookup.contains_key(&candidate) {
399            break candidate;
400        }
401    }
402}

404/// Maximum number of unpinned entries per relay in a [`RelayAddrMap`].
405///
406/// Bounding per relay ensures a relay forging source IDs can only evict unpinned entries
407/// from its own traffic.
408const MAX_UNPINNED_PER_RELAY: usize = 1024;
409
410/// Maximum number of relays with unpinned entries in a [`RelayAddrMap`].
411///
412/// Adding a relay beyond this limit evicts all unpinned entries of the least recently used
413/// relay, bounding entries left behind by relays we are no longer connected to.
414const MAX_UNPINNED_RELAYS: usize = 64;
415
416/// A bi-directional map between `(RelayUrl, EndpointId)` and [`RelayMappedAddr`].
417///
418/// Every datagram received over a relay needs a mapped address, and the relay chooses the
419/// source [`EndpointId`]. Noq may keep using a mapped address for as long as a connection
420/// or handshake refers to it, so entries are either pinned or unpinned:
421///
422/// - [`RelayAddrMap::get_unpinned`] looks up or inserts an entry for a received datagram.
423///   New entries are unpinned.
424///
425/// - The `RemoteStateActor` of an endpoint pins the entries of that endpoint which its
426///   connections use as paths, with [`RelayAddrMap::get_pinned`] or
427///   [`RelayAddrMap::pin_addr`]. Pinned entries are never evicted.
428///   [`RelayAddrMap::unpin_endpoint`] unpins them again once the actor stops.
429///
430/// - Unpinned entries are bounded per relay and in the number of relays. When a bound is
431///   exceeded, the oldest unpinned entries are removed.
432#[derive(Debug, Clone, Default)]
433pub(super) struct RelayAddrMap {
434    inner: Arc<std::sync::Mutex<RelayAddrMapInner>>,
435}
436
437impl RelayAddrMap {
438    /// Returns the [`RelayMappedAddr`] for `key`, pinning it for the endpoint it names.
439    pub(super) fn get_pinned(&self, key: &(RelayUrl, EndpointId)) -> RelayMappedAddr {
440        let mut inner = self.inner.lock().expect("poisoned");
441        if !inner.entries.addrs.contains_key(key) {
442            inner.entries.insert(key);
443        }
444        inner.entries.pin(key)
445    }
446
447    /// Pins the entry for `addr` and returns its key, if the entry exists and names
448    /// `endpoint_id`.
449    ///
450    /// Looking up and pinning under one lock ensures the entry cannot be evicted in between.
451    /// Entries naming another endpoint are not pinned and `None` is returned: they would
452    /// otherwise stay pinned for an endpoint whose actor may never run, and so never be
453    /// unpinned.
454    pub(super) fn pin_addr(
455        &self,
456        addr: &RelayMappedAddr,
457        endpoint_id: &EndpointId,
458    ) -> Option<(RelayUrl, EndpointId)> {
459        let mut inner = self.inner.lock().expect("poisoned");
460        let key = inner
461            .entries
462            .lookup
463            .get(addr)
464            .filter(|key| key.1 == *endpoint_id)
465            .cloned()?;
466        inner.entries.pin(&key);
467        Some(key)
468    }
469
470    /// Returns the [`RelayMappedAddr`] for `key`, inserting an unpinned entry if needed.
471    pub(super) fn get_unpinned(&self, key: &(RelayUrl, EndpointId)) -> RelayMappedAddr {
472        let mut guard = self.inner.lock().expect("poisoned");
473        let inner = &mut *guard;
474        // Only receive traffic refreshes a relay, so silent relays are evicted first.
475        let pushed = inner.touch_relay(&key.0);
476        match inner.entries.addrs.get(key) {
477            Some(RelayEntry {
478                addr,
479                queued: Some(seq),
480            }) => {
481                let (addr, seq) = (*addr, *seq);
482                if pushed - seq > MAX_UNPINNED_PER_RELAY as u64 / 2 {
483                    // Refresh active unpinned entries once they are past the midpoint of the
484                    // queue, avoiding a new queue record on every packet.
485                    inner.enqueue(key);
486                }
487                addr
488            }
489            Some(RelayEntry { addr, queued: None }) => *addr,
490            None => {
491                let addr = inner.entries.insert(key);
492                inner.enqueue(key);
493                addr
494            }
495        }
496    }
497
498    /// Unpins all entries of `endpoint_id` after its `RemoteStateActor` has stopped.
499    ///
500    /// The entries are queued rather than removed, so Noq can finish handshakes that are
501    /// still in flight and a reconnecting peer keeps its mapped address unless evicted in
502    /// the meantime.
503    pub(super) fn unpin_endpoint(&self, endpoint_id: &EndpointId) {
504        let mut inner = self.inner.lock().expect("poisoned");
505        let relays = inner.entries.pinned.remove(endpoint_id).unwrap_or_default();
506        for relay in relays {
507            inner.enqueue(&(relay, *endpoint_id));
508        }
509    }
510
511    /// Performs the reverse lookup.
512    pub(super) fn lookup(&self, addr: &RelayMappedAddr) -> Option<(RelayUrl, EndpointId)> {
513        let inner = self.inner.lock().expect("poisoned");
514        inner.entries.lookup.get(addr).cloned()
515    }
516
517    /// Returns the number of entries.
518    #[cfg(test)]
519    pub(super) fn len(&self) -> usize {
520        self.inner.lock().expect("poisoned").entries.addrs.len()
521    }
522
523    /// Returns whether an entry for `addr` exists and is pinned.
524    #[cfg(test)]
525    pub(super) fn is_pinned(&self, addr: &RelayMappedAddr) -> bool {
526        let inner = self.inner.lock().expect("poisoned");
527        inner
528            .entries
529            .lookup
530            .get(addr)
531            .and_then(|key| inner.entries.addrs.get(key))
532            .is_some_and(|entry| entry.queued.is_none())
533    }
534}

536#[derive(Debug, Default)]
537struct RelayAddrMapInner {
538    entries: RelayEntries,
539    /// Per-relay FIFO queues of unpinned entries.
540    queues: HashMap<RelayUrl, UnpinnedQueue>,
541    /// Monotonic counter incremented on [`RelayAddrMap::get_unpinned`], used for
542    /// [`UnpinnedQueue::last_used`].
543    tick: u64,
544}
545
546/// Forward, reverse, and pin tables for a [`RelayAddrMap`].
547#[derive(Debug, Default)]
548struct RelayEntries {
549    addrs: HashMap<(RelayUrl, EndpointId), RelayEntry>,
550    lookup: FxHashMap<RelayMappedAddr, (RelayUrl, EndpointId)>,
551    /// Relays on which each endpoint currently has pinned entries.
552    pinned: HashMap<EndpointId, HashSet<RelayUrl>>,
553}
554
555#[derive(Debug)]
556struct RelayEntry {
557    addr: RelayMappedAddr,
558    /// Sequence number of this entry's live record in its relay's [`UnpinnedQueue`], or
559    /// `None` if the entry is pinned.
560    ///
561    /// Queue records with any other sequence number are stale (left behind when the entry
562    /// was pinned or requeued) and are ignored on pop.
563    queued: Option<u64>,
564}
565
566#[derive(Debug, Default)]
567struct UnpinnedQueue {
568    /// Queued `(key, seq)` records from oldest to newest.
569    ///
570    /// May contain stale records; see [`RelayEntry::queued`].
571    records: VecDeque<((RelayUrl, EndpointId), u64)>,
572    /// Sequence number of the most recently pushed record.
573    pushed: u64,
574    /// Value of [`RelayAddrMapInner::tick`] when this relay last delivered a datagram.
575    last_used: u64,
576}
577
578impl RelayAddrMapInner {
579    /// Marks `relay` as recently used, first evicting the unpinned entries of the least
580    /// recently used relay if adding `relay` would exceed [`MAX_UNPINNED_RELAYS`].
581    ///
582    /// Returns [`UnpinnedQueue::pushed`] of `relay`'s queue.
583    fn touch_relay(&mut self, relay: &RelayUrl) -> u64 {
584        self.tick += 1;
585        // Runs on every received relay datagram, so the common case is a single lookup.
586        if let Some(queue) = self.queues.get_mut(relay) {
587            queue.last_used = self.tick;
588            return queue.pushed;
589        }
590        if self.queues.len() >= MAX_UNPINNED_RELAYS {
591            let least_recent = self
592                .queues
593                .iter()
594                .min_by_key(|(_, queue)| queue.last_used)
595                .map(|(relay, _)| relay.clone());
596            if let Some(queue) = least_recent.and_then(|relay| self.queues.remove(&relay)) {
597                for (key, seq) in queue.records {
598                    self.entries.evict_record(&key, seq);
599                }
600            }
601        }
602        let queue = UnpinnedQueue {
603            last_used: self.tick,
604            ..Default::default()
605        };
606        self.queues.insert(relay.clone(), queue);
607        0
608    }
609
610    /// Pushes `key` onto its relay's queue and evicts the oldest records past
611    /// [`MAX_UNPINNED_PER_RELAY`].
612    ///
613    /// Does not update `last_used`: if the relay has no queue yet, it is created with
614    /// `last_used = 0`, so unpinning entries of silent relays cannot displace active ones.
615    /// If [`MAX_UNPINNED_RELAYS`] queues already exist, the entry is evicted immediately.
616    fn enqueue(&mut self, key: &(RelayUrl, EndpointId)) {
617        let Self {
618            entries, queues, ..
619        } = self;
620        let Some(entry) = entries.addrs.get_mut(key) else {
621            return;
622        };
623        if !queues.contains_key(&key.0) && queues.len() >= MAX_UNPINNED_RELAYS {
624            entries.remove(key);
625            return;
626        }
627
628        let queue = queues.entry(key.0.clone()).or_default();
629        queue.pushed += 1;
630        queue.records.push_back((key.clone(), queue.pushed));
631        entry.queued = Some(queue.pushed);
632        while queue.records.len() > MAX_UNPINNED_PER_RELAY {
633            if let Some((key, seq)) = queue.records.pop_front() {
634                entries.evict_record(&key, seq);
635            }
636        }
637    }
638}
639
640impl RelayEntries {
641    /// Generates and inserts a fresh entry for `key` (which must not already exist).
642    ///
643    /// The caller must immediately either pin or enqueue the new entry.
644    fn insert(&mut self, key: &(RelayUrl, EndpointId)) -> RelayMappedAddr {
645        let addr = generate_unique(&self.lookup);
646        self.addrs
647            .insert(key.clone(), RelayEntry { addr, queued: None });
648        self.lookup.insert(addr, key.clone());
649        trace!(?addr, ?key, "generated new addr");
650        addr
651    }
652
653    /// Marks the existing entry for `key` as pinned by the endpoint it names.
654    fn pin(&mut self, key: &(RelayUrl, EndpointId)) -> RelayMappedAddr {
655        let entry = self.addrs.get_mut(key).expect("checked by caller");
656        entry.queued = None;
657        self.pinned.entry(key.1).or_default().insert(key.0.clone());
658        entry.addr
659    }
660
661    /// Removes the entry for `key` if `seq` is still its current queue record.
662    fn evict_record(&mut self, key: &(RelayUrl, EndpointId), seq: u64) {
663        if self
664            .addrs
665            .get(key)
666            .is_some_and(|entry| entry.queued == Some(seq))
667        {
668            self.remove(key);
669        }
670    }
671
672    /// Removes the entry for `key`.
673    fn remove(&mut self, key: &(RelayUrl, EndpointId)) {
674        if let Some(entry) = self.addrs.remove(key) {
675            self.lookup.remove(&entry.addr);
676            trace!(addr = ?entry.addr, ?key, "evicted unpinned addr");
677        }
678    }
679}

709    fn flood(map: &RelayAddrMap, relay: u8, n: u64) -> Vec<RelayMappedAddr> {

717    }
