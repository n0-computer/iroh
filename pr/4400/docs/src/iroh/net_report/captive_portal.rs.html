7use rand::seq::IndexedRandom;
8use tracing::trace;
9use url::Url;

11use super::https::http_client;

13/// Checks whether the host is behind a captive portal.
14///
15/// Returns `true` if we appear to be behind one. The check requests a
16/// relay's `/generate_204` endpoint: a portal intercepting HTTP answers with
17/// something other than the expected empty 204.
18pub(super) async fn check(
19    dns_resolver: DnsResolver,
20    relay_map: RelayMap,
21    preferred_relay: Option<RelayUrl>,
22    tls_config: rustls::ClientConfig,
23    proxy_url: Option<Url>,
24) -> Result<bool> {
25    trace!("captive portal check starting");
26    // Use the preferred relay if it is still in the map, otherwise a random one.
27    let url = match preferred_relay.filter(|url| relay_map.contains(url)) {
28        Some(url) => url,
29        None => {
30            let urls: Vec<_> = relay_map.urls();
31            let Some(url) = urls.choose(&mut rand::rng()) else {
32                trace!("No suitable relay for captive portal check");
33                return Ok(false);
34            };
35            url.clone()
36        }
37    };
38
39    let client = http_client(dns_resolver, tls_config, proxy_url)?;
40
41    // Note: the set of valid characters in a challenge and the total
42    // length is limited; see is_challenge_char in bin/iroh-relay for more
43    // details.
44
45    let host_name = url.host_str().unwrap_or_default();
46    let challenge = format!("ts_{host_name}");
47    let portal_url = format!("http://{host_name}/generate_204");
48    let res = client
49        .request(reqwest::Method::GET, portal_url)
50        .header("X-Iroh-Challenge", &challenge)
51        .send()
52        .await
53        .std_context("Captive portal HTTP request failed")?;
54
55    let expected_response = format!("response {challenge}");
56    let is_valid_response = res
57        .headers()
58        .get("X-Iroh-Response")
59        .map(|s| s.to_str().unwrap_or_default())
60        == Some(&expected_response);
61
62    trace!(
63        "check_captive_portal url={} status_code={} valid_response={}",
64        res.url(),
65        res.status(),
66        is_valid_response,
67    );
68    let has_captive = res.status() != 204 || !is_valid_response;
69
70    Ok(has_captive)
71}
