1//! Timeouts and pacing of the net report.
2
3use n0_future::time::Duration;
4
5/// Seconds a cycle waits for a second relay before it picks the home relay.
6///
7/// It is public for use in documentation and in the timeout calculations of
8/// callers.
9pub const FIRST_REPORT_TIMEOUT_SECS: u64 = 3;
10
11/// Timeout for the captive portal HTTP request itself.
12pub(super) const CAPTIVE_PORTAL_TIMEOUT: Duration = Duration::from_secs(2);
13
14/// Timeouts and pacing of the net report.
15///
16/// A probe cycle starts on a trigger: startup, a network change, a change of
17/// the relay map, or the periodic refresh. It ends when every probe has
18/// finished or [`Self::cycle_timeout`] has passed. These values are not
19/// configurable by users yet. Tests shorten them.
20#[derive(Debug, Clone)]
21pub(crate) struct Timing {
22    /// Minimum time between the starts of two probe cycles.
23    ///
24    /// A trigger within this window after a cycle started waits until the
25    /// window ends, and further triggers merge into it, so a burst of network
26    /// changes runs a single cycle. A network change also waits this long
27    /// before its cycle starts, so that the socket has rebound when the probes
28    /// go out.
29    pub(crate) request_debounce: Duration,
30    /// Maximum wait for a second relay before a cycle picks the home relay.
31    pub(crate) first_report_timeout: Duration,
32    /// Timeout of a single QAD or HTTPS probe.
33    ///
34    /// Longer than the first-report deadline, so that a probe on a degraded
35    /// link still completes and refines what the cycle knows.
36    pub(crate) probe_timeout: Duration,
37    /// Maximum duration of a probe cycle.
38    ///
39    /// Bounds the network activity of a cycle, including probes delayed by
40    /// the QAD stagger.
41    pub(crate) cycle_timeout: Duration,
42    /// Age of the last full cycle after which the next cycle is full.
43    pub(crate) full_cycle_interval: Duration,
44    /// Delay before the captive portal check starts.
45    ///
46    /// Gives fast QAD probes time to succeed first: if UDP works, a QAD answer
47    /// cancels the captive portal check before it starts.
48    pub(crate) captive_portal_delay: Duration,
49    /// Time between two refresh cycles, before a random jitter of up to 30%.
50    ///
51    /// The default gives 20 to 26 seconds, just under the common UDP NAT
52    /// timeout of 30 seconds.
53    pub(crate) refresh_interval: Duration,
54}
55
56impl Default for Timing {
57    fn default() -> Self {
58        Self {
59            request_debounce: Duration::from_millis(100),
60            first_report_timeout: Duration::from_secs(FIRST_REPORT_TIMEOUT_SECS),
61            probe_timeout: Duration::from_secs(15),
62            cycle_timeout: Duration::from_secs(30),
63            full_cycle_interval: Duration::from_secs(5 * 60),
64            captive_portal_delay: Duration::from_millis(200),
65            refresh_interval: Duration::from_secs(20),
66        }
67    }
68}
