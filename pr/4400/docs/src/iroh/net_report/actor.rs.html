1//! The actor that probes the relays and publishes the [`Outputs`].
2//!
3//! [`NetReportActor`] runs in the background for as long as the endpoint is
4//! alive. It learns about the network in three ways:
5//!
6//! - QAD (QUIC Address Discovery) opens a QUIC connection to a relay. The
7//!   relay reports the public address it sees us coming from, and the round
8//!   trip measures our latency to that relay.
9//! - An HTTPS probe measures latency to a relay with a plain GET request. It
10//!   finds no address, but it still works on networks that block QUIC.
11//! - The captive portal check looks for a network that intercepts HTTP.
12//!
13//! The actor is a single loop that waits for these events:
14//!
15//! - The network interfaces changed.
16//! - The relay map changed.
17//! - The relay transport's connection to the home relay changed.
18//! - A probe finished.
19//! - A kept QAD connection reported an address or closed.
20//! - A timer fired.
21//!
22//! Probes are futures the loop polls itself, and kept QAD connections are
23//! streams it polls itself, so the actor spawns no tasks and owns everything
24//! it runs. Dropping a probe or a connection stops it.
25//!
26//! QAD connections are long-lived, and that shapes the rest. For each address
27//! family the actor races a probe to several relays and keeps the connection
28//! that answers first. A second answer decides whether our address varies by
29//! destination, and then the actor stops the others. The kept connection
30//! stays open, and the relay reports a change of our address on it within a
31//! keep-alive interval. When a kept connection closes, the actor runs a
32//! refresh to replace it.
33//!
34//! A probe cycle is a round of probing. A [`ProbeScope::Full`] cycle starts
35//! from nothing: a major network change and a change of the relay map run
36//! one, and so does the periodic full cycle. Everything else runs a
37//! [`ProbeScope::Refresh`] that keeps what still works. The actor publishes a
38//! public address as soon as a relay reports it. It decides the home relay
39//! twice per cycle: early, once two relays have answered and the current home
40//! relay with them, or at [`Timing::first_report_timeout`] when they are slow,
41//! and again when the cycle ends. Only completed failures can disqualify an
42//! existing home relay; a connected relay transport vouches for it.
43//!
44//! [`Timing::first_report_timeout`]: super::timing::Timing::first_report_timeout

95///
96/// The variants order `Refresh < Full` so that merging two pending triggers

99enum ProbeScope {
100    /// Keeps the open QAD connections and probes only the missing families.
101    ///
102    /// It also re-picks the home relay. It measures HTTPS latency when the

106    /// Closes the open QAD connections and runs every probe from scratch.
107    ///
108    /// This is what a real network change calls for. Once the debounce has
109    /// passed, a `Full` trigger aborts a running cycle.
110    Full,

131/// Returns the next update from a kept QAD connection.
132fn next_qad_event(
133    conns: &mut PerFamily<Option<QadConn>>,
134) -> impl Future<Output = (AddrFamily, QadConnEvent)> + '_ {
135    future::poll_fn(|cx| {
136        for family in AddrFamily::ALL {
137            if let Some(conn) = &mut conns[family]
138                && let Poll::Ready(Some(event)) = conn.poll_next(cx)
139            {
140                return Poll::Ready((family, event));
141            }
142        }
143        Poll::Pending
144    })
145}

158/// Runs a probe after `delay`, under `timeout`, until `cancel` fires.
159///
160/// Errors and timeouts establish failure. Cancellation and panics do not
161/// establish reachability. It logs failures in the caller's span. The delay sits
162/// outside the timeout, so a staggered probe still gets its full timeout once
163/// it starts. The actor polls probes itself, so catching a panic here keeps a
164/// bug in one probe from stopping the actor.
165async fn guarded<T, E: fmt::Display>(
166    cancel: CancellationToken,
167    delay: Duration,
168    timeout: Duration,
169    work: impl Future<Output = Result<T, E>>,
170) -> ProbeResult<T> {
171    let result = cancel
172        .run_until_cancelled(async {
173            if !delay.is_zero() {
174                time::sleep(delay).await;
175            }
176            time::timeout(timeout, AssertUnwindSafe(work).catch_unwind()).await
177        })
178        .await;
179    match result {
180        Some(Ok(Ok(Ok(value)))) => ProbeResult::Success(value),
181        Some(Ok(Ok(Err(err)))) => {
182            debug!(err = %format_args!("{err:#}"), "probe failed");
183            ProbeResult::Failed
184        }
185        Some(Ok(Err(_panic))) => {
186            error!("probe panicked");
187            ProbeResult::Inconclusive
188        }
189        Some(Err(_elapsed)) => {
190            debug!("probe timed out");
191            ProbeResult::Failed
192        }
193        None => ProbeResult::Inconclusive,
194    }
195}

202/// Our public address in one family, as far as we know it.
203#[derive(Debug, Clone, Copy, PartialEq, Eq)]
204struct Known {
205    /// The address and port the relays see us coming from.
206    addr: SocketAddr,
207    /// Whether the address differs per destination, `None` while undecided.
208    varies: Option<bool>,
209}
210
211impl Known {
212    /// Returns what we know after a kept connection reported `addr`.
213    ///
214    /// A change on a single connection means our address changed, for
215    /// example because the NAT rebound. It does not decide whether the
216    /// address varies by destination, so the verdict stays.
217    fn observe(known: Option<Self>, addr: SocketAddr) -> Self {
218        Self {
219            addr,
220            varies: known.and_then(|k| k.varies),
221        }
222    }
223
224    /// Returns the published form of the address, if it is of type `A`.
225    fn public_addr<A>(
226        self,
227        matching: impl FnOnce(SocketAddr) -> Option<A>,
228    ) -> Option<PublicAddr<A>> {
229        Some(PublicAddr {
230            addr: matching(self.addr)?,
231            varies_by_dest: self.varies == Some(true),
232        })
233    }
234}

236/// How far the running cycle got with one address family.
237///
238/// The family's [`Known`] address stays what the last cycle established.
239/// [`Progress::view`] and [`Progress::settle`] combine the two into what to
240/// publish while the cycle runs, and what remains when it ends.
241#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
242enum Progress {
243    /// A connection kept from an earlier cycle vouches for the family.
244    ///
245    /// The cycle does not probe it, and the connection keeps the known
246    /// address current.
247    Kept,
248    /// No relay has answered for the family in this cycle.
249    #[default]
250    Waiting,
251    /// One relay answered with this address.
252    One(SocketAddr),
253    /// At least two relays answered, so the cycle decided the verdict.
254    Decided {
255        /// The address of the first answer.
256        addr: SocketAddr,
257        /// Whether any later answer saw a different address.
258        varies: bool,
259    },
260}
261
262impl Progress {
263    /// Returns the progress after another relay answered with `addr`.
264    fn answer(self, addr: SocketAddr) -> Self {
265        match self {
266            Self::Kept | Self::Waiting => Self::One(addr),
267            Self::One(first) => Self::Decided {
268                addr: first,
269                varies: first != addr,
270            },
271            Self::Decided {
272                addr: first,
273                varies,
274            } => Self::Decided {
275                addr: first,
276                varies: varies || first != addr,
277            },
278        }
279    }
280
281    /// Returns the progress after a connection kept in this cycle saw `addr`.
282    fn observe(self, addr: SocketAddr) -> Self {
283        match self {
284            Self::One(_) => Self::One(addr),
285            Self::Decided { varies, .. } => Self::Decided { addr, varies },
286            other => other,
287        }
288    }
289
290    /// Returns what to publish for the family while the cycle runs.
291    ///
292    /// The known address stays until the cycle learns a new one. A known
293    /// verdict carries over only to the same address.
294    fn view(self, known: Option<Known>) -> Option<Known> {
295        match self {
296            Self::Kept | Self::Waiting => known,
297            Self::One(addr) => Some(Known {
298                addr,
299                varies: known.filter(|k| k.addr == addr).and_then(|k| k.varies),
300            }),
301            Self::Decided { addr, varies } => Some(Known {
302                addr,
303                varies: Some(varies),
304            }),
305        }
306    }
307
308    /// Returns what we know about the family once the cycle ended.
309    ///
310    /// Like [`Self::view`], except that a family no relay answered for loses
311    /// its address: nothing confirmed it.
312    fn settle(self, known: Option<Known>) -> Option<Known> {
313        match self {
314            Self::Waiting => None,
315            _ => self.view(known),
316        }
317    }
318}

320/// Where a home relay decision looks for candidates.
321#[derive(Debug, Clone, Copy, PartialEq, Eq)]
322enum Candidates {
323    /// Only relays the cycle measured.
324    ///
325    /// A cycle that heard from no relay, for example while we are offline,

327    Measured,
328    /// Relays the cycle measured, or else relays with a latency of the last
329    /// five minutes.

331    /// Used when the relay transport could not dial the home relay. That
332    /// happens between cycles, when the cycle may not have measured another
333    /// relay.
334    Recent,
335}

337/// How the current home relay compares at the end of a cycle.
338#[derive(Debug, Clone, Copy, PartialEq, Eq)]
339enum CurrentHome {
340    /// The cycle measured its latency.
341    Measured,
342    /// No fresh result establishes reachability, for example after cancellation.
343    ///
344    /// Its latency of the last five minutes remains the hysteresis baseline.
345    Unknown,
346    /// Completed probes failed while its transport was disconnected, the
347    /// relay transport failed to dial it, or it left the relay map.
348    Unreachable,
349}

351/// Latencies of recent cycles, for picking the home relay.
352#[derive(Debug, Default)]
353struct History {
354    /// Latencies of the cycles of the last five minutes, by end time.
355    prev: BTreeMap<Instant, RelayLatencies>,
356    /// Start of the last `Full` cycle, or `None` before the first one.
357    last_full: Option<Instant>,
358}
359
360impl History {
361    /// Maximum age of recorded latencies considered for relay selection.
362    const MAX_AGE: Duration = Duration::from_secs(5 * 60);
363
364    /// Records the latencies of a finished cycle.
365    fn record(&mut self, latencies: &RelayLatencies, now: Instant) {
366        self.prev
367            .retain(|t, _| now.duration_since(*t) <= Self::MAX_AGE);
368        self.prev.insert(now, latencies.clone());
369    }
370
371    /// Returns whether the last finished cycle measured no latency at all.
372    fn last_was_empty(&self) -> bool {
373        self.prev
374            .last_key_value()
375            .is_none_or(|(_, latencies)| latencies.is_empty())
376    }
377
378    /// Picks the home relay among the relays in `latencies`.
379    ///
380    /// The candidate is the relay in `relay_map` with the best latency across
381    /// the last five minutes. It replaces the current home relay only when it
382    /// is at least 33% faster, or when the current one is
383    /// [`CurrentHome::Unreachable`]. An unreachable home relay is no
384    /// candidate. Returns `None`, which keeps the current home relay, when
385    /// there is no candidate.
386    fn select_home_relay(
387        &self,
388        latencies: &RelayLatencies,
389        current: Option<(&RelayUrl, CurrentHome)>,
390        candidates: Candidates,
391        relay_map: &RelayMap,
392        now: Instant,
393    ) -> Option<RelayUrl> {
394        // Best latency per relay over the last five minutes.
395        let mut best_recent = RelayLatencies::default();
396        for (t, prev) in self.prev.iter() {
397            if now.duration_since(*t) <= Self::MAX_AGE {
398                best_recent.merge(prev);
399            }
400        }
401        best_recent.merge(latencies);
402
403        let unreachable = current
404            .filter(|(_, status)| *status == CurrentHome::Unreachable)
405            .map(|(url, _)| url);
406        let best_of = |set: &RelayLatencies| {
407            set.iter()
408                .map(|(_, url, _)| url)
409                .filter(|url| relay_map.contains(url) && Some(*url) != unreachable)
410                .filter_map(|url| Some((url.clone(), best_recent.get(url)?)))
411                .min_by_key(|(_, latency)| *latency)
412        };
413        let (best_url, best_latency) = match candidates {
414            Candidates::Measured => best_of(latencies),
415            Candidates::Recent => best_of(latencies).or_else(|| best_of(&best_recent)),
416        }?;
417        let Some((current, status)) = current else {
418            return Some(best_url);
419        };
420        let current_latency = match status {
421            CurrentHome::Measured => latencies.get(current),
422            CurrentHome::Unknown => best_recent.get(current),
423            CurrentHome::Unreachable => None,
424        };
425
426        // Hysteresis: don't switch if the new relay isn't much better.
427        if let Some(current_latency) = current_latency
428            && !current_latency.is_zero()
429            && best_latency > current_latency / 3 * 2
430        {
431            return Some(current.clone());
432        }
433        Some(best_url)
434    }
435
436    /// Decides the scope of the next cycle.
437    ///
438    /// A `Full` trigger always yields a `Full` cycle. A `Refresh` trigger is
439    /// promoted to `Full` on the first cycle, once `full_cycle_interval` has
440    /// elapsed, or when we are behind a captive portal without UDP.
441    fn scope_for(
442        &self,
443        trigger: ProbeScope,
444        now: Instant,
445        full_cycle_interval: Duration,
446        captive_without_udp: bool,
447    ) -> ProbeScope {
448        let full = trigger == ProbeScope::Full
449            || self
450                .last_full
451                .is_none_or(|t| now.duration_since(t) > full_cycle_interval)
452            || captive_without_udp;
453        if full {
454            ProbeScope::Full
455        } else {
456            ProbeScope::Refresh
457        }
458    }
459}

590    /// Runs the actor until the shutdown token is cancelled.
591    ///
592    /// On shutdown, dropping `self` drops the cycle and the kept QAD
593    /// connections, which stops every probe.
594    pub(super) async fn run(mut self) {
595        let mut interfaces_closed = false;
596        let mut relay_status_closed = false;
597        self.maybe_start(Instant::now());
598        loop {
599            let timer = time::sleep_until(self.next_deadline());
600            n0_future::pin!(timer);
601
602            tokio::select! {
603                biased;
604
605                _ = self.shutdown.cancelled() => break,
606                change = self.interfaces.updated(), if !interfaces_closed => match change {
607                    Ok(is_major) => self.on_network_change(is_major),
608                    Err(_) => interfaces_closed = true,
609                },
610                Some(message) = self.messages.recv() => match message {
611                    Message::RelaysChanged => {
612                        debug!("relay map changed");
613                        self.trigger(ProbeScope::Full);
614                    }
615                    Message::NetworkChange => self.interfaces.network_change().await,
616                    #[cfg(all(test, with_crypto_provider))]
617                    Message::ForceNetworkChange(is_major) => self.on_network_change(is_major),
618                },
619                status = self.opts.relay_status.updated(), if !relay_status_closed => match status {
620                    Ok(status) => self.on_relay_status(status),
621                    Err(_) => relay_status_closed = true,
622                },
623                Some(output) = next_probe_output(&mut self.cycle) => self.on_probe_output(output),
624                event = next_qad_event(&mut self.qad_conns) => self.on_qad_event(event),
625                _ = &mut timer => self.on_timer(Instant::now()),
626            }
627        }
628    }

630    /// Publishes a network change and waits for a usable route before notifying QUIC.
631    fn on_network_change(&mut self, is_major: bool) {
632        #[cfg(not(wasm_browser))]
633        if is_major {
634            self.opts.dns_resolver.reset();
635        }
636        self.interfaces
637            .publish_change(is_major, &self.outputs.network_state);
638        // The socket rebinds on the same change. Wait out the debounce, so
639        // the probes go out on the new sockets.
640        self.not_before = Some(Instant::now() + self.opts.timing.request_debounce);
641        self.trigger(if is_major {
642            ProbeScope::Full
643        } else {
644            ProbeScope::Refresh
645        });
646    }

648    /// Handles a change of the relay transport's home relay connection.
649    ///
650    /// When the relay transport could not dial the home relay, the home relay
651    /// is unreachable, and the actor decides without it right away. It does
652    /// so once until fresh evidence of connectivity arrives.
653    fn on_relay_status(&mut self, status: Option<RelayStatus>) {
654        let Some(status) = status else {
655            return;
656        };
657        if Some(status.url()) != self.outputs.home_relay.get().as_ref() {
658            return;
659        }
660        if status.is_connected() {
661            self.failed_over = false;
662            return;
663        }
664        if self.relay_unreachable(status.url()) {
665            debug!(relay_url = %status.url(), "home relay unreachable, deciding without it");
666            if self.decide_home_relay(self.cycle.as_ref(), Candidates::Recent) {
667                self.failed_over = true;
668                self.publish();
669            }
670        }
671    }

697    /// Returns the earliest instant at which [`Self::on_timer`] has work.
698    fn next_deadline(&self) -> Instant {
699        // A pending cycle waits for the debounce, unless it waits for the
700        // running cycle to end.
701        let debounce = self
702            .pending
703            .filter(|&scope| self.cycle.is_none() || scope == ProbeScope::Full)
704            .and_then(|_| self.start_allowed_at());
705        let cycle = self.cycle.as_ref();
706        [
707            debounce,
708            self.interfaces.next_deadline(),
709            cycle.and_then(|c| c.decide_at),
710            cycle.map(|c| c.deadline),
711        ]
712        .into_iter()
713        .flatten()
714        .fold(self.next_refresh, Instant::min)
715    }

717    /// Returns when a pending cycle may start, or `None` if it may start now.
718    ///
719    /// That is once the request debounce after the last cycle start has
720    /// passed, and the debounce after the last network change.
721    fn start_allowed_at(&self) -> Option<Instant> {
722        let debounce = self
723            .last_cycle_start
724            .map(|t| t + self.opts.timing.request_debounce);
725        debounce.max(self.not_before)
726    }

728    /// Handles every deadline that has passed by `now`.
729    fn on_timer(&mut self, now: Instant) {
730        self.interfaces.on_timer(now, &self.outputs.network_state);
731        if self
732            .cycle
733            .as_ref()
734            .and_then(|c| c.decide_at)
735            .is_some_and(|t| t <= now)
736        {
737            debug!("first report deadline passed, deciding the home relay");
738            self.decide_home_relay_early(true);
739        }
740        if self.cycle.as_ref().is_some_and(|c| c.deadline <= now) {
741            debug!("cycle timeout passed, ending cycle");
742            self.finish_cycle(now);
743        }
744        if self.next_refresh <= now {
745            self.next_refresh = now + jittered(self.opts.timing.refresh_interval);
746            self.pending = Some(self.pending.unwrap_or(ProbeScope::Refresh));
747        }
748        self.maybe_start(now);
749    }

751    /// Starts the pending cycle, if it may start now.
752    ///
753    /// A pending cycle waits for the request debounce. After that a `Full`
754    /// cycle aborts a running cycle, while a `Refresh` waits until the running
755    /// cycle ends.
756    fn maybe_start(&mut self, now: Instant) {
757        let Some(scope) = self.pending else {
758            return;
759        };
760        if self.start_allowed_at().is_some_and(|t| now < t) {
761            return;
762        }
763        if self.cycle.is_some() {
764            if scope == ProbeScope::Refresh {
765                trace!("deferring refresh until the running cycle ends");
766                return;
767            }
768            debug!("aborting running cycle");
769            self.cycle = None;
770        }
771        self.pending = None;
772        self.start_cycle(scope, now);
773    }

775    /// Starts a new probe cycle.
776    fn start_cycle(&mut self, trigger: ProbeScope, now: Instant) {
777        self.last_cycle_start = Some(now);
778        self.not_before = None;
779        self.next_refresh = now + jittered(self.opts.timing.refresh_interval);
780        if self.opts.relay_map.is_empty() {
781            debug!("empty relay map, clearing net_report state");
782            self.clear();
783            return;
784        }
785
786        let has_udp = AddrFamily::ALL.into_iter().any(|f| self.known[f].is_some());
787        let scope = self.history.scope_for(
788            trigger,
789            now,
790            self.opts.timing.full_cycle_interval,
791            !has_udp && self.captive_portal == Some(true),
792        );
793        debug!(?trigger, ?scope, "starting probe cycle");
794        if scope == ProbeScope::Full {
795            self.close_qad_conns();
796            self.captive_portal = None;
797            self.history.last_full = Some(now);
798            self.metrics.reports_full.inc();
799        }
800        self.metrics.reports.inc();
801
802        // Measure latencies from scratch, starting with the kept QAD
803        // connections. Drop a connection that closed before we handled its
804        // close event, so the cycle probes its family. A `Full` cycle forgets
805        // a family whose interface is gone.
806        self.latencies = RelayLatencies::default();
807        let mut progress = PerFamily::<Progress>::default();
808        for family in AddrFamily::ALL {
809            if self.qad_conns[family]
810                .as_ref()
811                .is_some_and(QadConn::is_closed)
812            {
813                self.qad_conns[family] = None;
814            }
815            if scope == ProbeScope::Full && !self.interfaces.has_family(family) {
816                self.known[family] = None;
817            }
818            if let Some(report) = self.qad_conns[family]
819                .as_ref()
820                .and_then(QadConn::current_report)
821            {
822                progress[family] = Progress::Kept;
823                self.known[family] = Some(Known::observe(self.known[family], report.addr));
824                self.latencies
825                    .update_relay(report.relay_url, report.latency, family.probe());
826            }
827        }
828
829        let mut cycle = ProbeCycle {
830            probes: FuturesUnordered::new(),
831            decide_at: Some(now + self.opts.timing.first_report_timeout),
832            deadline: now + self.opts.timing.cycle_timeout,
833            probed: BTreeSet::new(),
834            answered: BTreeSet::new(),
835            failed: BTreeSet::new(),
836            https_cancel: BTreeMap::new(),
837            progress,
838            qad_cancel: PerFamily::default(),
839            captive_portal_cancel: CancellationToken::new(),
840        };
841        #[cfg(not(wasm_browser))]
842        {
843            self.add_qad_probes(&mut cycle);
844            if scope == ProbeScope::Full && self.opts.config.captive_portal_check {
845                self.add_captive_portal_check(&mut cycle);
846            }
847        }
848        // A full cycle measures every relay over HTTPS, and so does a cycle
849        // after one that measured nothing. A refresh only checks the home
850        // relay, and only when the relay transport has no healthy connection
851        // to it. A kept QAD connection does not vouch for it: it may be dead
852        // without knowing it yet.
853        let https_relays = if scope == ProbeScope::Full || self.history.last_was_empty() {
854            self.opts.relay_map.relays()
855        } else {
856            self.outputs
857                .home_relay
858                .get()
859                .filter(|url| !self.relay_connected(url))
860                .and_then(|url| self.opts.relay_map.get(&url))
861                .into_iter()
862                .collect()
863        };
864        self.add_https_probes(&mut cycle, https_relays);
865
866        let done = cycle.probes.is_empty();
867        self.cycle = Some(cycle);
868        self.publish();
869        // A cycle can start with no probes at all: every family has a kept
870        // connection and HTTPS is not due. It is complete right away.
871        if done {
872            self.finish_cycle(now);
873        }
874    }

889    /// Applies one probe result.
890    fn on_probe_output(&mut self, output: ProbeOutput) {
891        let Some(cycle) = &mut self.cycle else {
892            return;
893        };
894        match output {
895            ProbeOutput::Qad(family, _, ProbeResult::Success((report, conn))) => {
896                debug!(?family, ?report, "QAD probe completed");
897                cycle.cancel_https(&report.relay_url);
898                // UDP works, so skip captive portal detection.
899                cycle.captive_portal_cancel.cancel();
900                cycle.answered.insert(report.relay_url.clone());
901                self.failed_over = false;
902                let progress = &mut cycle.progress[family];
903                *progress = progress.answer(report.addr);
904                if let Progress::Decided { varies: true, .. } = progress {
905                    debug!(?family, "public address varies by destination");
906                }
907                self.latencies
908                    .update_relay(report.relay_url, report.latency, family.probe());
909                if self.qad_conns[family].is_none() {
910                    // Keep this connection, and let the other probes run so a
911                    // second answer can decide mapping-varies. The address may
912                    // have changed since the probe returned.
913                    if let Some(fresh) = conn.current_report() {
914                        *progress = progress.observe(fresh.addr);
915                    }
916                    self.qad_conns[family] = Some(conn);
917                } else {
918                    // The second answer decided mapping-varies, so stop the
919                    // family's remaining probes and drop this connection.
920                    conn.close();
921                    cycle.qad_cancel[family].cancel();
922                }
923            }
924            ProbeOutput::Https(_, ProbeResult::Success(report)) => {
925                debug!(?report, "HTTPS probe completed");
926                cycle.cancel_https(&report.relay_url);
927                cycle.answered.insert(report.relay_url.clone());
928                self.failed_over = false;
929                self.latencies
930                    .update_relay(report.relay_url, report.latency, Probe::Https);
931            }
932            ProbeOutput::CaptivePortal(ProbeResult::Success(found)) => {
933                debug!(found, "captive portal check completed");
934                self.captive_portal = Some(found);
935            }
936            ProbeOutput::Qad(_, url, ProbeResult::Failed)
937            | ProbeOutput::Https(url, ProbeResult::Failed) => {
938                cycle.failed.insert(url);
939            }
940            // Cancellation, a panic, or a failed captive portal check tells us nothing about a relay.
941            _ => {}
942        }
943        let done = cycle.probes.is_empty();
944        self.decide_home_relay_early(false);
945        self.publish();
946        if done {
947            self.finish_cycle(Instant::now());
948        }
949    }

953    /// A connection kept in the running cycle updates the cycle's progress.
954    /// One kept from an earlier cycle updates the known address directly.

961                *progress = progress.observe(report.addr);
962            }
963            _ => self.known[family] = Some(Known::observe(self.known[family], report.addr)),
964        }

992    /// Decides the home relay early, if the cycle has enough answers.
993    ///
994    /// That is once two relays have answered (or the only one, with a single
995    /// relay), and the current home relay has answered if the cycle probes
996    /// it. When `deadline_passed`, the cycle decides with what it has.
997    fn decide_home_relay_early(&mut self, deadline_passed: bool) {
998        let Some(cycle) = &self.cycle else {
999            return;
1000        };
1001        if cycle.decide_at.is_none() {
1002            return;
1003        }
1004        let home_pending = self
1005            .outputs
1006            .home_relay
1007            .get()
1008            .is_some_and(|url| cycle.probed.contains(&url) && !cycle.answered.contains(&url));
1009        let enough = cycle.answered.len() >= 2.min(self.opts.relay_map.len());
1010        if !deadline_passed && (home_pending || !enough) {
1011            return;
1012        }
1013        let changed = self.decide_home_relay(Some(cycle), Candidates::Measured);
1014        self.failed_over |= changed && cycle.answered.is_empty();
1015        if let Some(cycle) = &mut self.cycle {
1016            cycle.decide_at = None;
1017        }
1018    }

1020    /// Picks the home relay with what `cycle` learned so far.
1021    ///
1022    /// Without a cycle, it decides with the latencies of the last one.
1023    fn decide_home_relay(&self, cycle: Option<&ProbeCycle>, candidates: Candidates) -> bool {
1024        let current = self.outputs.home_relay.get();
1025        // Apply the offline guard to cycle decisions as well as transport failures.
1026        // A relay removed from the map must still be replaced.
1027        if self.failed_over
1028            && current
1029                .as_ref()
1030                .is_some_and(|url| self.opts.relay_map.contains(url))
1031        {
1032            return false;
1033        }
1034        let current = current.as_ref().map(|url| {
1035            let status = if !self.opts.relay_map.contains(url) || self.relay_unreachable(url) {
1036                CurrentHome::Unreachable
1037            } else if cycle.is_some_and(|c| c.answered.contains(url)) {
1038                CurrentHome::Measured
1039            } else if !self.relay_connected(url)
1040                && cycle.is_some_and(|c| c.probes.is_empty() && c.failed.contains(url))
1041            {
1042                CurrentHome::Unreachable
1043            } else {
1044                CurrentHome::Unknown
1045            };
1046            (url, status)
1047        });
1048        let selected = self.history.select_home_relay(
1049            &self.latencies,
1050            current,
1051            candidates,
1052            &self.opts.relay_map,
1053            Instant::now(),
1054        );
1055        self.set_home_relay(selected)
1056    }

1113    /// Ends the running cycle.
1114    ///
1115    /// Drops the cycle's remaining probes, drops what the cycle could not
1116    /// confirm, picks the home relay, and records the latencies. Then starts a
1117    /// cycle that waited for this one.
1118    fn finish_cycle(&mut self, now: Instant) {
1119        let Some(mut cycle) = self.cycle.take() else {
1120            return;
1121        };
1122        // Stop unfinished probes; only failures already collected count as evidence.
1123        cycle.probes = FuturesUnordered::new();
1124        for family in AddrFamily::ALL {
1125            let progress = cycle.progress[family];
1126            if progress == Progress::Waiting && self.known[family].is_some() {
1127                debug!(?family, "dropping public address the cycle did not confirm");
1128            }
1129            self.known[family] = progress.settle(self.known[family]);
1130        }
1131        let changed = self.decide_home_relay(Some(&cycle), Candidates::Measured);
1132        self.failed_over |= changed && cycle.answered.is_empty();
1133        self.history.record(&self.latencies, now);
1134        self.publish();
1135        debug!(
1136            latencies = ?self.latencies,
1137            duration = ?self.last_cycle_start.map(|t| now.duration_since(t)),
1138            "net_report cycle complete",
1139        );
1140        self.maybe_start(now);
1141    }

1152    /// Adds a QAD probe for each relay and each family that needs one.
1153    ///
1154    /// A family needs probes when it has no kept connection, its interface is
1155    /// up, and the options enable it.
1156    #[cfg(not(wasm_browser))]
1157    fn add_qad_probes(&self, cycle: &mut ProbeCycle) {
1158        let Some(quic_client) = &self.opts.quic_client else {
1159            return;
1160        };
1161        let needed: Vec<AddrFamily> = AddrFamily::ALL
1162            .into_iter()
1163            .filter(|&family| {
1164                self.qad_conns[family].is_none()
1165                    && self.interfaces.has_family(family)
1166                    && (family == AddrFamily::V4 || self.opts.qad_ipv6)
1167            })
1168            .collect();
1169
1170        // One probe per (relay, needed family). Stagger successive probes by
1171        // an increasing delay so they do not all start at once.
1172        let stagger = self.opts.config.qad_stagger;
1173        let mut index = 0u32;
1174        for relay in self
1175            .opts
1176            .relay_map
1177            .relays::<Vec<_>>()
1178            .into_iter()
1179            .take(MAX_QAD_RELAYS)
1180        {
1181            for &family in &needed {
1182                cycle.probed.insert(relay.url.clone());
1183                let work = qad::run_probe(
1184                    family,
1185                    relay.clone(),
1186                    quic_client.clone(),
1187                    self.opts.dns_resolver.clone(),
1188                );
1189                let probe = guarded(
1190                    cycle.qad_cancel[family].clone(),
1191                    stagger * index,
1192                    self.opts.timing.probe_timeout,
1193                    work,
1194                );
1195                let span = info_span!("QAD", ?family, relay_url = %relay.url);
1196                let relay_url = relay.url.clone();
1197                cycle.probes.push(Box::pin(
1198                    async move { ProbeOutput::Qad(family, relay_url, probe.await) }
1199                        .instrument(span),
1200                ));
1201                index += 1;
1202            }
1203        }
1204    }

1241    /// Adds the captive portal check.
1242    ///
1243    /// Delayed by [`Timing::captive_portal_delay`] to give QAD probes time to
1244    /// succeed first, and cancelled if QAD confirms UDP connectivity.
1245    ///
1246    /// [`Timing::captive_portal_delay`]: super::timing::Timing::captive_portal_delay
1247    #[cfg(not(wasm_browser))]
1248    fn add_captive_portal_check(&self, cycle: &mut ProbeCycle) {
