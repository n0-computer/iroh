19/// Configuration for the portmapper service (UPnP, PCP, NAT-PMP).
20///
21/// Port mapping asks the local router to open an external port so peers can
22/// reach this endpoint directly, improving connectivity behind NATs. The
23/// discovery step (UPnP uses SSDP multicast) can, however, trigger firewall
24/// prompts on some networks — see [`PortmapperConfig::Disabled`].
25///
26/// Used with [`crate::endpoint::Builder::portmapper_config`].
27#[derive(Debug, Clone)]
28#[non_exhaustive]
29pub enum PortmapperConfig {
30    /// Enable portmapping with default settings.
31    ///
32    /// This is the default.
33    #[non_exhaustive]
34    Enabled {},
35    /// Disable portmapping.
36    ///
37    /// Skips the UPnP/PCP/NAT-PMP gateway probing. Use this to avoid the
38    /// SSDP multicast discovery that can raise firewall dialogs (notably on
39    /// macOS), at the cost of potentially worse direct connectivity behind
40    /// some NATs.
41    Disabled,
42}
43
44impl Default for PortmapperConfig {
45    fn default() -> Self {
46        PortmapperConfig::Enabled {}
47    }
48}
49
50pub(crate) fn create_client(
51    config: &PortmapperConfig,
52    metrics: EndpointMetrics,
53    shutdown: CancellationToken,
54) -> Client {
55    #[cfg(all(not(wasm_browser), feature = "portmapper"))]
56    if matches!(config, PortmapperConfig::Enabled { .. }) {
57        let service = ::portmapper::Client::default();
58        let external = service.watch_external_address();
59        let interval = rand::rng().random_range(Duration::from_secs(20)..=Duration::from_secs(26));
60        return Client::Enabled {
61            service,
62            external,
63            retry: time::interval(interval),
64            metrics,
65            shutdown,
66        };
67    }
68    let _ = (config, metrics, shutdown);
69    Client::Disabled
70}

72/// Owns mapping retries, address notifications, and mapping release on drop.
73#[derive(Debug)]
74pub(crate) enum Client {
75    #[cfg(all(not(wasm_browser), feature = "portmapper"))]
76    Enabled {
77        service: ::portmapper::Client,
78        external: watch::Receiver<Option<SocketAddrV4>>,
79        retry: time::Interval,
80        metrics: EndpointMetrics,
81        shutdown: CancellationToken,
82    },
83    Disabled,
84}
85
86impl Client {
87    /// Waits for a changed mapping while retrying failed acquisition attempts.
88    pub(crate) async fn updated(&mut self) -> Option<SocketAddrV4> {
89        #[cfg(all(not(wasm_browser), feature = "portmapper"))]
90        while let Self::Enabled {
91            external,
92            retry,
93            metrics,
94            ..
95        } = self
96        {
97            tokio::select! {
98                changed = external.changed() => {
99                    if changed.is_err() { break; }
100                    let external = *external.borrow_and_update();
101                    metrics.socket.actor_tick_portmap_changed.inc();
102                    if external.is_some() {
103                        metrics.net_report.portmap_external_address_updated.inc();
104                    }
105                    debug!(?external, "external address updated");
106                    return external;
107                }
108                _ = retry.tick() => {
109                    metrics.socket.actor_tick_re_stun.inc();
110                }
111            }
112            self.procure_mapping();
113        }
114        std::future::pending().await
115    }
116
117    pub(crate) fn external_address(&self) -> Option<SocketAddrV4> {
118        match self {
119            #[cfg(all(not(wasm_browser), feature = "portmapper"))]
120            Self::Enabled { external, .. } => *external.borrow(),
121            Self::Disabled => None,
122        }
123    }
124
125    /// Asks for a mapping if none exists; the service renews existing mappings.
126    pub(crate) fn procure_mapping(&self) {
127        #[cfg(all(not(wasm_browser), feature = "portmapper"))]
128        if let Self::Enabled {
129            service,
130            metrics,
131            shutdown,
132            ..
133        } = self
134        {
135            if shutdown.is_cancelled() {
136                trace!("skipping port mapping: shutting down");
137                return;
138            }
139            metrics.net_report.portmap_attempts.inc();
140            service.procure_mapping();
141        }
142    }
143
144    pub(crate) fn update_local_port(&self, _port: std::num::NonZeroU16) {
145        #[cfg(all(not(wasm_browser), feature = "portmapper"))]
146        if let Self::Enabled { service, .. } = self {
147            service.update_local_port(_port);
148        }
149    }
150}
151
152impl Drop for Client {
153    fn drop(&mut self) {
154        #[cfg(all(not(wasm_browser), feature = "portmapper"))]
155        if let Self::Enabled { service, .. } = self {
156            service.deactivate();
157        }
158    }
159}
