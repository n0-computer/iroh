1//! Learns our public addresses and the best relay, and keeps both current.
2//!
3//! Net report probes the configured relays. QUIC address discovery (QAD) tells
4//! us the public address the relays see us coming from, and the round trips
5//! tell us which relay is closest. The net report re-assesses both when the
6//! network changes and periodically.
7//!
8//! It runs on its own: it watches the network interfaces itself and needs no
9//! requests. What it learns it publishes as [`Outputs`], small watchable values
10//! the rest of iroh subscribes to. The socket watches the public addresses, and
11//! the relay transport watches the home relay. The full report is only a
12//! debug view, exposed through `Endpoint::net_report`.
13// Based on <https://github.com/tailscale/tailscale/blob/main/net/netcheck/netcheck.go>

66/// Configuration for the net report component.
67///
68/// Controls which probes and checks the net report performs.
69#[derive(Debug, Clone)]
70#[non_exhaustive]
71pub struct NetReportConfig {
72    /// Whether to run HTTPS latency probes against relay servers.
73    ///
74    /// HTTPS latency probes perform an empty HTTPS GET request to each configured
75    /// relay server and measure latency.
76    ///
77    /// They are performed in addition to the QUIC address discovery (QAD) probes.
78    /// In networks that do not allow QUIC traffic, they are the only way to detect
79    /// relay latencies and thus the preferred relay.
80    ///
81    /// Disabling them is harmless on networks that do allow QUIC traffic, but will
82    /// completely prevent finding the home relay on networks that do block QUIC.
83    pub https_probes: bool,
84
85    /// Whether to check for captive portals.
86    ///
87    /// This is done by accessing a well-known URL that is available on each relay
88    /// server, `/generate_204`. If a GET request to this URL returns anything else
89    /// but a 204 No Content response, we assume we are behind a captive portal.
90    ///
91    /// The check runs on every full probe cycle: at startup, after a major
92    /// network change, and periodically. When we have detected that we are
93    /// behind a captive portal and UDP does not work, every cycle is a full
94    /// one, in case the captive portal status changes.
95    pub captive_portal_check: bool,
96
97    /// Delay inserted between successive QAD probes within a probe cycle.
98    ///
99    /// A cycle runs QUIC address discovery against several relays. By default
100    /// those probes all start at once, a brief burst of QUIC handshakes. On
101    /// resource-constrained devices (an ESP32, say) that burst is disruptive,
102    /// so a non-zero value spreads the probes out: the first still starts
103    /// immediately, and each following one is delayed by a further multiple
104    /// of this duration. Setting it does not delay the first result.
105    ///
106    /// Defaults to [`Duration::ZERO`], which keeps the all-at-once behavior.
107    pub qad_stagger: Duration,
108}
109
110impl NetReportConfig {
111    /// Creates a minimal configuration that disables all optional probes and checks.
112    pub fn minimal() -> Self {
113        Self {
114            https_probes: false,
115            captive_portal_check: false,
116            qad_stagger: Duration::ZERO,
117        }
118    }
119}
120
121impl Default for NetReportConfig {
122    fn default() -> Self {
123        Self {
124            https_probes: true,
125            captive_portal_check: true,
126            qad_stagger: Duration::ZERO,
127        }
128    }
129}

170impl Builder {
171    pub(crate) fn new() -> Self {

175    /// Read-only access for the relay transport and the report API.
176    pub(crate) fn outputs(&self) -> Outputs {
177        self.outputs.clone()

180    /// Starts interface monitoring and probing, returning the owning client.
181    pub(crate) async fn spawn(
182        self,
183        opts: Options,
184        metrics: Arc<Metrics>,
185        shutdown: CancellationToken,
186    ) -> Result<Client, netwatch::netmon::Error> {
187        let interfaces = Interfaces::new().await?;
188        Ok(self.spawn_with_interfaces(opts, interfaces, metrics, shutdown))
189    }
190
191    fn spawn_with_interfaces(
192        self,
193        opts: Options,
194        interfaces: Interfaces,
195        metrics: Arc<Metrics>,
196        shutdown: CancellationToken,
197    ) -> Client {
198        let network = self.outputs.network_state.watch();
199        let public_addrs = self.outputs.public_addrs.watch();
200        let (messages, receiver) = mpsc::unbounded_channel();
201        let actor =
202            NetReportActor::new(opts, interfaces, receiver, self.outputs, metrics, shutdown);
203        let handle = task::spawn(
204            actor
205                .run()
206                .instrument(info_span!(parent: Span::current(), "net_report")),
207        );
208        Client {
209            messages,
210            network,
211            public_addrs,
212            changes: Changes::default(),
213            ready: Changes::default(),
214            _actor_handle: AbortOnDropHandle::new(handle),
215        }
216    }
217}

219/// Owns the net report task, its commands, and the socket's output subscription.
220///
221/// Dropping the client stops the actor and its probes. Cloning the outputs
222/// does not extend the task's lifetime.
223#[derive(Debug)]
224pub(crate) struct Client {
225    /// Changes the actor cannot observe itself.
226    messages: mpsc::UnboundedSender<Message>,
227    /// The addresses and network changes consumed by the socket.
228    network: Direct<NetworkState>,
229    public_addrs: Direct<PublicAddrs>,
230    /// Last consumed network changes, including any coalesced major changes.
231    changes: Changes,
232    ready: Changes,
233    /// Aborts the background actor when the client is dropped.
234    _actor_handle: AbortOnDropHandle<()>,
235}
236
237#[derive(Debug)]
238enum Message {
239    RelaysChanged,
240    NetworkChange,
241    #[cfg(all(test, with_crypto_provider))]
242    ForceNetworkChange(bool),
243}
244
245impl Client {
246    /// Waits for the next address or network change.
247    pub(crate) async fn updated(&mut self) -> Update {
248        loop {
249            let network = self.network.get();
250            if let Some(is_major) = network.changes.since(self.changes) {
251                self.changes = network.changes;
252                return Update::NetworkChanged { is_major };
253            }
254            if let Some(ready) = network.ready
255                && let Some(is_major) = ready.since(self.ready)
256            {
257                self.ready = ready;
258                return Update::NetworkReady {
259                    is_major,
260                    hint: network.hint(),
261                };
262            }
263            tokio::select! {
264                Ok(_) = self.network.updated() => {},
265                Ok(_) = self.public_addrs.updated() => return Update::AddressesChanged,
266                else => std::future::pending().await,
267            }
268        }
269    }
270
271    /// The current public addresses, without consuming update notifications.
272    #[cfg(not(wasm_browser))]
273    pub(crate) fn public_addrs(&self) -> PublicAddrs {
274        self.public_addrs.clone().get()
275    }
276
277    /// Expands bound sockets into usable local addresses.
278    #[cfg(not(wasm_browser))]
279    pub(crate) fn local_addresses(
280        &self,
281        bound_local: &[(SocketAddr, SocketAddr)],
282        include_loopback: bool,
283    ) -> Vec<SocketAddr> {
284        self.network
285            .clone()
286            .get()
287            .local_addresses(bound_local, include_loopback)
288    }
289
290    /// Asks the interface monitor to check for a network change.
291    pub(crate) fn network_change(&self) {
292        self.messages.send(Message::NetworkChange).ok();
293    }
294
295    #[cfg(all(test, with_crypto_provider))]
296    pub(crate) fn force_network_change(&self, is_major: bool) {
297        self.messages
298            .send(Message::ForceNetworkChange(is_major))
299            .ok();
300    }
301
302    /// Tells the net report that the relay map changed, starting a full cycle.
303    pub(crate) fn relays_changed(&self) {
304        self.messages.send(Message::RelaysChanged).ok();
305    }
306}

314    pub(super) async fn relay() -> (server::Server, RelayConfig) {
315        let server = server::Server::spawn(server::testing::server_config())
316            .await
317            .expect("should serve relay");
318        let quic = Some(RelayQuicConfig::new(
319            server.quic_addr().expect("server should run quic").port(),
320        ));
321        let endpoint_desc =
322            RelayConfig::new(server.https_url().expect("should work as relay"), quic);
323
324        (server, endpoint_desc)
325    }
326
327    /// Create a [`crate::RelayMap`] of the given size.
328    pub(super) async fn relay_map(relays: usize) -> (Vec<server::Server>, crate::RelayMap) {
329        let mut servers = Vec::with_capacity(relays);
330        let mut endpoints = Vec::with_capacity(relays);
331        for _ in 0..relays {
332            let (relay_server, endpoint) = relay().await;
333            servers.push(relay_server);
334            endpoints.push(endpoint);
335        }
336        (servers, crate::RelayMap::from_iter(endpoints))
337    }

392                    .expect("infallible"),

395            let interfaces = Watchable::new(netwatch::netmon::State::fake());
396            let builder = Builder::new();
397            let outputs = builder.outputs();
398            let metrics = Arc::new(Metrics::default());
399            let cancel = CancellationToken::new();
400            let client = builder.spawn_with_interfaces(
401                opts,
402                Interfaces::from_watcher(interfaces.watch()),
403                metrics.clone(),
404                cancel.child_token(),

407                client,
408                outputs,
409                interfaces,
410                metrics,
411                ep,
412                cancel,

416        /// Waits until `watcher` holds a value that satisfies `f`, and returns it.
417        async fn wait<W: Watcher>(
418            &self,
419            mut watcher: W,
420            f: impl Fn(&W::Value) -> bool,
421        ) -> W::Value {
422            time::timeout(TEST_TIMEOUT, async {
423                loop {
424                    let value = watcher.get();
425                    if f(&value) {
426                        return value;
427                    }
428                    watcher.updated().await.expect("watcher closed");
429                }
430            })
431            .await
432            .expect("timed out waiting for output")
433        }
434
435        /// Waits for the actor to process an interface change without consuming client updates.
436        async fn set_interfaces(&self, state: netwatch::netmon::State) {
437            let previous = self.outputs.network_state.get().changes;
438            self.interfaces.set(state).unwrap();
439            self.wait(self.outputs.network_state.watch(), |network| {
440                network.changes != previous
441            })
442            .await;
443        }
444
445        /// Waits until `n` cycles have started.
446        async fn wait_cycles(&self, n: u64) {
447            time::timeout(TEST_TIMEOUT, async {
448                while self.metrics.reports.get() < n {
449                    time::sleep(Duration::from_millis(10)).await;
450                }
451            })
452            .await
453            .expect("timed out waiting for cycles");
454        }
455
456        /// Stops the net report and waits for its endpoint to close.
457        async fn shutdown(self) {
458            self.cancel.cancel();
459            drop(self.client);
460            self.ep.wait_idle().await;
461        }
462    }

471    /// Readiness for an older usable network must not bypass a newer route wait.
472    #[tokio::test(start_paused = true)]
473    async fn test_network_readiness_ignores_superseded_ready_state() -> Result<()> {
474        let mut client = TestClient::new(RelayMap::empty())?;
475        assert!(client.client.updated().now_or_never().is_none());
476
477        let mut state = client.interfaces.get();
478        state.is_expensive = true;
479        client.set_interfaces(state.clone()).await;
480        state.default_route_interface = None;
481        client.set_interfaces(state).await;
482
483        assert!(matches!(
484            client.client.updated().await,
485            Update::NetworkChanged { is_major: true }
486        ));
487        assert!(client.client.updated().now_or_never().is_none());
488        tokio::time::advance(Duration::from_secs(4)).await;
489        assert!(client.client.updated().now_or_never().is_none());
490        // A minor change must neither restart the deadline nor forget the major one.
491        let previous = client.outputs.network_state.get().changes;
492        client.client.force_network_change(false);
493        client
494            .wait(client.outputs.network_state.watch(), |network| {
495                network.changes != previous
496            })
497            .await;
498        assert!(matches!(
499            client.client.updated().await,
500            Update::NetworkChanged { is_major: false }
501        ));
502        assert!(client.client.updated().now_or_never().is_none());
503        tokio::time::advance(Duration::from_secs(1)).await;
504        assert!(matches!(
505            client.client.updated().await,
506            Update::NetworkReady { is_major: true, .. }
507        ));
508        assert!(client.client.updated().now_or_never().is_none());
509        client.shutdown().await;
510        Ok(())
511    }

572    /// A periodic refresh reuses the kept connections.
573    #[tokio::test(flavor = "multi_thread")]
574    #[traced_test]
575    async fn test_refresh_after_full() -> Result<()> {
576        let (servers, relay_map) = test_utils::relay_map(2).await;
577        let timing = timing::Timing {
578            refresh_interval: Duration::from_millis(300),
579            ..Default::default()
580        };
581        let client = TestClient::with_timing(relay_map, timing)?;
582        let first = client
583            .wait(client.outputs.report(), has_v4_and_relay)
584            .await
585            .unwrap();
586        assert_eq!(
587            first
588                .relay_latency
589                .iter()
590                .map(|(_, url, _)| url)
591                .collect::<std::collections::BTreeSet<_>>()
592                .len(),
593            2,
594        );
595
596        client.wait_cycles(2).await;
597        assert_eq!(client.metrics.reports_full.get(), 1);
598        // The kept connection still provides the address.
599        let addrs = client.outputs.public_addrs().get();
600        assert_eq!(addrs.v4.map(|a| a.addr), first.global_v4);
601        assert!(client.outputs.home_relay().get().is_some());
602
603        client.shutdown().await;
604        for server in servers {
605            server.shutdown().await?;
606        }
607        Ok(())
608    }
