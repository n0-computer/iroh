91    pub(super) fn poll_recv(
92        &mut self,
93        cx: &mut Context,
94        bufs: &mut [io::IoSliceMut<'_>],
95        metas: &mut [noq_udp::RecvMeta],
96        recv_infos: &mut [RecvInfo],
97    ) -> Poll<io::Result<usize>> {
98        assert_eq!(bufs.len(), metas.len(), "non matching bufs & metas");
99        assert_eq!(
100            bufs.len(),
101            recv_infos.len(),
102            "non matching bufs & recv_infos"
103        );
104        let mut num_msgs = 0;
105        // Bound the work even if every received batch is dropped. Only advance the
106        // output slot when we actually fill it.
107        for _ in 0..bufs.len() {
108            let buf_out = &mut bufs[num_msgs];
109            let meta_out = &mut metas[num_msgs];
110            let recv_info = &mut recv_infos[num_msgs];
111            let dm = match self.poll_recv_queue(cx) {
112                Poll::Ready(Some(recv)) => recv,
113                Poll::Ready(None) => {
114                    error!("relay_recv_channel closed");
115                    return Poll::Ready(Err(io::Error::new(
116                        io::ErrorKind::NotConnected,
117                        "connection closed",
118                    )));
119                }
120                Poll::Pending => {
121                    if num_msgs == 0 {
122                        return Poll::Pending;
123                    }
124                    break;
125                }
126            };
127
128            // This *tries* to make the datagrams fit into our buffer by re-batching them.
129            let num_segments = dm
130                .datagrams
131                .segment_size
132                .map_or(1, |ss| buf_out.len() / u16::from(ss) as usize);
133            let datagrams = dm.datagrams.take_segments(num_segments);
134            let empty_now = datagrams.contents.is_empty();
135            let empty_after = dm.datagrams.contents.is_empty();
136
137            let dm = RelayRecvDatagram {
138                datagrams,
139                src: dm.src,
140                url: dm.url.clone(),
141            };
142
143            // If `take_segments` processed the whole contents (empty_after) or none at all
144            // (empty_now) we shouldn't process what's left in `self.pending_item`.
145            // In the first case the remaining `pending_item` is empty and can be dropped.
146            // In the second case a single segment could not fit into the buffer, future
147            // calls to `poll_recv` would not change this so drop the `pending_item` with
148            // the oversized segment size.
149            if empty_after || empty_now {
150                self.pending_item = None;
151            }
152            if empty_now {
153                warn!(
154                    noq_buf_len = buf_out.len(),
155                    segment_size = ?dm.datagrams.segment_size,
156                    "dropping received datagram: segment_size too large");
157                continue;
158            }
159
160            if buf_out.len() < dm.datagrams.contents.len() {
161                // Our receive buffer isn't big enough to process this datagram.
162                // Copying it would cause a panic.
163                warn!(
164                    noq_buf_len = buf_out.len(),
165                    datagram_len = dm.datagrams.contents.len(),
166                    segment_size = ?dm.datagrams.segment_size,
167                    "dropping received datagram: noq buffer too small"
168                );
169                // In theory we could put some logic in here to fragment the datagram in case
170                // we still have enough room in our `buf_out` left to fit a couple of
171                // `dm.datagrams.segment_size`es, but we *should* have cut those datagrams
172                // to appropriate sizes earlier in the pipeline (just before we put them
173                // into the `relay_datagram_recv_queue` in the `ActiveRelayActor`).
174                // So the only case in which this happens is we receive a datagram via the relay
175                // that's essentially bigger than our configured `max_udp_payload_size`.
176                // In that case we drop it and let MTU discovery take over.
177                continue;
178            }
179
180            buf_out[..dm.datagrams.contents.len()].copy_from_slice(&dm.datagrams.contents);
181            meta_out.len = dm.datagrams.contents.len();
182            meta_out.stride = dm
183                .datagrams
184                .segment_size
185                .map_or(dm.datagrams.contents.len(), |s| u16::from(s) as usize);
186            meta_out.ecn = None;
187            meta_out.dst_ip = None;
188
189            *recv_info = RecvInfo::from_addr((dm.url, dm.src).into());
190            num_msgs += 1;
191        }
192
193        // If we have any msgs to report, they are in the first `num_msgs` slots.
194        if num_msgs > 0 {
195            assert!(num_msgs <= metas.len());
196            Poll::Ready(Ok(num_msgs))
197        } else {
198            if !bufs.is_empty() {
199                // Every receive attempt dropped data, so we haven't polled the queue
200                // to Pending. More data may still be queued, and no queue wake is
201                // guaranteed. Wake to ensure another poll.
202                cx.waker().wake_by_ref();
203            }
204            Poll::Pending
205        }
206    }

225    /// Makes sure we have a pending item stored, if not, it'll poll a new one from the queue.
226    ///
227    /// Returns a mutable reference to the stored pending item.
228    #[inline]
229    fn poll_recv_queue<'a>(
230        &'a mut self,
231        cx: &mut Context,
232    ) -> Poll<Option<&'a mut RelayRecvDatagram>> {
233        // Borrow checker doesn't quite understand an if let Some(_)... here
234        if self.pending_item.is_some() {
235            return Poll::Ready(self.pending_item.as_mut());
236        }
237
238        let item = match self.relay_datagram_recv_queue.poll_recv(cx) {
239            Poll::Ready(Some(item)) => item,
240            Poll::Ready(None) => return Poll::Ready(None),
241            Poll::Pending => return Poll::Pending,
242        };
243
244        Poll::Ready(Some(self.pending_item.insert(item)))
245    }
246}

259    pub(super) fn rebind(&self) -> io::Result<()> {
260        self.send_relay_actor(RelayActorMessage::MaybeCloseRelaysOnRebind);
261
262        Ok(())
263    }
264
265    fn send_relay_actor(&self, msg: RelayActorMessage) {
266        match self.sender.try_send(msg) {
267            Ok(_) => {}
268            Err(mpsc::error::TrySendError::Closed(_)) => {
269                warn!("unable to send to relay actor, already closed");
270            }
271            Err(mpsc::error::TrySendError::Full(_)) => {
272                warn!("dropping message for relay actor, channel is full");
273            }
274        }
275    }

286impl RelaySender {
287    pub(super) fn is_valid_send_addr(&self, _url: &RelayUrl, _endpoint_id: &EndpointId) -> bool {
288        true
289    }
290
291    pub(super) fn poll_send(
292        &mut self,
293        cx: &mut Context,
294        dest_url: RelayUrl,
295        dest_endpoint: EndpointId,
296        transmit: &Transmit<'_>,
297    ) -> Poll<io::Result<()>> {
298        match ready!(self.sender.poll_reserve(cx)) {
299            Ok(()) => {
300                let contents = datagrams_from_transmit(transmit);
301                let item = RelaySendItem {
302                    remote_endpoint: dest_endpoint,
303                    url: dest_url.clone(),
304                    datagrams: contents,
305                };
306                match self.sender.send_item(item) {
307                    Ok(()) => Poll::Ready(Ok(())),
308                    Err(_err) => Poll::Ready(Err(io::Error::new(
309                        io::ErrorKind::ConnectionReset,
310                        "channel to actor is closed",
311                    ))),
312                }
313            }
314            Err(_err) => Poll::Ready(Err(io::Error::new(
315                io::ErrorKind::ConnectionReset,
316                "channel to actor is closed",
317            ))),
318        }
319    }
320}

322/// Translate a UDP transmit to the `Datagrams` type for sending over the relay.
323fn datagrams_from_transmit(transmit: &Transmit<'_>) -> Datagrams {
324    Datagrams {
325        ecn: transmit.ecn.map(|ecn| match ecn {
326            noq_udp::EcnCodepoint::Ect0 => noq_proto::EcnCodepoint::Ect0,
327            noq_udp::EcnCodepoint::Ect1 => noq_proto::EcnCodepoint::Ect1,
328            noq_udp::EcnCodepoint::Ce => noq_proto::EcnCodepoint::Ce,
329        }),
330        segment_size: transmit
331            .segment_size
332            .map(|ss| ss as u16)
333            .and_then(NonZeroU16::new),
334        contents: Bytes::copy_from_slice(transmit.contents),
335    }
336}

378        let capacity = 16;

381
382        let mut tasks = JoinSet::new();
383
384        tasks.spawn({
385            async move {
386                let mut expected_msgs: BTreeSet<usize> = (0..capacity).collect();
387                while !expected_msgs.is_empty() {
388                    let datagram: RelayRecvDatagram = receiver.recv().await.unwrap();
389                    let msg_num = usize::from_le_bytes(datagram.datagrams.contents.as_ref().try_into().unwrap());
390                    debug!("Received {msg_num}");

394                    }
395                }
396            }
397        });

399        for i in 0..capacity {
400            tasks.spawn({
401                let sender = sender.clone();

403                async move {

405                    sender

409                            datagrams: Datagrams::from(&i.to_le_bytes()),
410                        })

412                }

551                    .is_pending()
552            );

554            assert_eq!(transport.relay_datagram_recv_queue.len(), 1);
